    79	    session: AsyncSession, tenant_id: uuid.UUID, order_id: uuid.UUID
    80	) -> SalesOrder:
    81	    order = await session.get(SalesOrder, order_id)

   141	def _validate_billing_line(
   142	    order_id: uuid.UUID,
   143	    order_lines: dict[uuid.UUID, SalesOrderLine],
   144	    payload_line: object,
   145	    currency_code: str,
   146	) -> _BillingLineInput:
   147	    """Validate one billing line against its order line: the line belongs to the order, the qty is
   148	    > 0 and within the still-open-to-invoice quantity = delivered − invoiced (over-billing REJECTED
   149	    422 ``sales.over_billing`` — you cannot invoice beyond what shipped)."""
   150	    order_line = order_lines.get(payload_line.sales_order_line_id)  # type: ignore[attr-defined]
   151	    if order_line is None:
   152	        raise ValidationFailedError(
   153	            message="The billing line does not belong to this sales order",
   154	            code="sales.billing_line_not_on_order",
   155	            details={
   156	                "sales_order_id": str(order_id),
   157	                "sales_order_line_id": str(
   158	                    payload_line.sales_order_line_id  # type: ignore[attr-defined]
   159	                ),
   160	            },
   161	        )
   162	    qty = _validate_quantity(payload_line.quantity)  # type: ignore[attr-defined]
   163	    open_qty = Decimal(str(order_line.delivered_quantity)) - Decimal(
   164	        str(order_line.invoiced_quantity)
   165	    )
   166	    if qty > open_qty:
   167	        raise ValidationFailedError(
   168	            message="The billed quantity exceeds the delivered-not-yet-invoiced quantity",
   169	            code="sales.over_billing",
   170	            details={
   171	                "sales_order_line_id": str(order_line.id),
   172	                "open_quantity": str(open_qty),
   173	                "quantity": str(qty),
   174	            },
   175	        )
   176	    return _build_billing_line(
   177	        order_line,
   178	        delivery_line_id=payload_line.delivery_line_id,  # type: ignore[attr-defined]
   179	        qty=qty,
   180	        currency_code=currency_code,
   181	    )

   184	def _build_all_delivered_lines(
   185	    order_lines: dict[uuid.UUID, SalesOrderLine], currency_code: str
   186	) -> list[_BillingLineInput]:
   187	    """The convenience path: bill every order line's full delivered-not-yet-invoiced quantity (lines
   188	    with nothing open are skipped). No delivery line is named (the chain is order → billing)."""
   189	    built: list[_BillingLineInput] = []
   190	    for order_line in order_lines.values():
   191	        open_qty = Decimal(str(order_line.delivered_quantity)) - Decimal(
   192	            str(order_line.invoiced_quantity)
   193	        )
   194	        if open_qty <= 0:
   195	            continue
   196	        built.append(
   197	            _build_billing_line(
   198	                order_line, delivery_line_id=None, qty=open_qty, currency_code=currency_code
   199	            )
   200	        )
   201	    return built

   204	async def create_billing(
   205	    session: AsyncSession, tenant_id: uuid.UUID, payload: BillingCreate
   206	) -> SalesBilling:
   207	    """Create a DRAFT billing against a sales order (PLAN 7.4). Validates the order is at least
   208	    partially delivered (else 422 ``sales.order_not_delivered``), each line belongs to it and does
   209	    not over-bill its delivered-not-invoiced quantity (over-billing → 422 ``sales.over_billing``).
   210	    Snapshots the customer + payment terms from the order and the priced fields per line from the
   211	    order line, claims the BIL number (D-040), and links each named delivery → billing. No journal
   212	    yet — that is POST. ``bill_all_delivered`` bills every delivered-not-invoiced line."""
   213	    order = await _require_billable_order(session, tenant_id, payload.sales_order_id)
   214	    order_lines = {
   215	        line.id: line for line in await get_sales_order_lines(session, tenant_id, order.id)
   216	    }
   217	    billing_date = payload.billing_date or date.today()
   218	
   219	    if payload.bill_all_delivered:
   220	        validated = _build_all_delivered_lines(order_lines, order.currency_code)
   221	    else:
   222	        if not payload.lines:
   223	            raise ValidationFailedError(
   224	                message="A billing needs at least one line (or bill_all_delivered)",
   225	                code="sales.billing_no_lines",
   226	            )
   227	        validated = [
   228	            _validate_billing_line(order.id, order_lines, line, order.currency_code)
   229	            for line in payload.lines
   230	        ]
   231	    if not validated:
   232	        raise ValidationFailedError(
   233	            message="The order has no delivered-not-yet-invoiced quantity to bill",
   234	            code="sales.nothing_to_bill",
   235	            details={"sales_order_id": str(order.id)},
   236	        )
   237	
   238	    total = sum((line.line_amount for line in validated), Decimal(0))
   239	    billing_id = uuid.uuid4()
   240	    document = await docflow.register_document(
   241	        session,
   242	        tenant_id,
   243	        BILLING_DOC_TYPE,
   244	        billing_id,
   245	        doc_number=None,
   246	        status=BillingStatus.DRAFT.value,
   247	    )
   248	    number = await claim_document_number(
   249	        session,
   250	        tenant_id,
   251	        sequence_name=BILLING_SEQUENCE_NAME,
   252	        prefix=BILLING_NUMBER_PREFIX,
   253	        padding=BILLING_NUMBER_PADDING,
   254	        on_date=billing_date,
   255	    )
   256	    billing = SalesBilling(
   257	        id=billing_id,
   258	        tenant_id=tenant_id,
   259	        document_id=document.id,
   260	        billing_number=number,
   261	        status=BillingStatus.DRAFT.value,
   262	        sales_order_id=order.id,
   263	        customer_id=order.customer_id,
   264	        currency_code=order.currency_code,
   265	        billing_date=billing_date,
   266	        payment_terms_days=order.payment_terms_days,
   267	        total_amount=total,
   268	        notes=payload.notes,
   269	    )
   270	    session.add(billing)
   271	    for index, line in enumerate(validated, start=1):
   272	        session.add(
   273	            SalesBillingLine(
   274	                tenant_id=tenant_id,
   275	                billing_id=billing_id,
   276	                line_number=index,
   277	                sales_order_line_id=line.sales_order_line_id,
   278	                delivery_line_id=line.delivery_line_id,
   279	                item_id=line.item_id,
   280	                quantity=line.quantity,
   281	                unit_price=line.unit_price,
   282	                discount_type=line.discount_type,
   283	                discount_value=line.discount_value,
   284	                line_amount=line.line_amount,
   285	                tax_code_id=line.tax_code_id,
   286	            )
   287	        )
   288	    await session.flush()
   289	    await docflow.set_document_status(
   290	        session, tenant_id, document.id, doc_number=number, status=BillingStatus.DRAFT.value
   291	    )
   292	    await _link_delivery_predecessors(session, tenant_id, billing_id, validated)
   293	    return billing

   298	    tenant_id: uuid.UUID,
   299	    billing_id: uuid.UUID,
   300	    lines: list[_BillingLineInput],

   302	    """Link each distinct source delivery → billing ('invoiced_by') for the docflow chain (PLAN

   329	async def cancel_billing(
   330	    session: AsyncSession, tenant_id: uuid.UUID, billing_id: uuid.UUID
   331	) -> SalesBilling:
   332	    """Cancel a DRAFT billing (PLAN 7.4). A POSTED billing is TERMINAL — it triggered an AR invoice
   333	    (recognized revenue + AR), so it is corrected by a return / credit note (7.4), never cancelled.
   334	    Cancelling a DRAFT raises no invoice and changes no invoiced_quantity."""
   335	    billing = await get_billing(session, tenant_id, billing_id)
   336	    if BillingStatus(billing.status) != BillingStatus.DRAFT:
   337	        raise ConflictError(
   338	            message=f"A {billing.status} billing cannot be cancelled",
   339	            code="sales.billing_not_cancellable",
   340	            details={"billing_id": str(billing_id), "status": billing.status},
   341	        )
   342	    billing.status = BillingStatus.CANCELLED.value
   343	    await session.flush()
   344	    await docflow.set_document_status(
   345	        session, tenant_id, billing.document_id, status=BillingStatus.CANCELLED.value
   346	    )
   347	    return billing

     2	(STRUCTURE §8.4, the delivery_post precedent). The DRAFT create + helpers + ``cancel_billing`` stay
     3	in ``billing.py``; the reads in ``billing_reads.py``; the post path + its post-only helpers live
     4	here. ``__init__`` re-exports all three as one ``service`` surface (``service.post_billing``).
     5	
     6	``post_billing`` is the heart: in ONE transaction it PUBLISHES ``BillingInvoiced`` — finance's
     7	handler
     8	creates + posts the AR customer invoice (Dr AR control / Cr sales-revenue + Cr output tax) — raises
     9	each order line's invoiced_quantity, advances the order status (INVOICED / CLOSED), links docflow
    10	order → billing, and sets the billing POSTED. A closed billing period trips the AR invoice's journal
    11	trigger and rolls the WHOLE post back.
    12	
    13	The MIRROR of procurement's ``post_invoice_match``: sales resolves the AR control + sales-revenue
    14	accounts (+ the customer's payment terms snapshot for the due date) from finance/queries UP FRONT
    15	(downward reads — a missing posting default fails the post before any state change, D-046), then
    16	publishes the event with those accounts so finance's handler is a thin builder. Sales never imports

    60	async def post_billing(
    61	    session: AsyncSession, tenant_id: uuid.UUID, billing_id: uuid.UUID
    62	) -> SalesBilling:
    63	    """Post a DRAFT billing (PLAN 7.4, D-046) — the heart. In ONE transaction: PUBLISH
    64	    ``BillingInvoiced`` so finance's handler creates + posts the AR customer invoice (Dr AR control
    65	    /
    66	    Cr sales-revenue + Cr output tax, partner_id = customer id, due = billing_date + terms); raise
    67	    each order line's invoiced_quantity; advance the order (INVOICED when fully invoiced, CLOSED
    68	    when
    69	    fully delivered AND invoiced); link docflow order → billing; set the billing POSTED. A closed
    70	    billing period trips the AR invoice's journal trigger and rolls it ALL back.
    71	
    72	    Only a DRAFT billing posts; a POSTED one is idempotent-rejected (terminal). The caller commits
    73	    via
    74	    uow; the published event drains in the same uow."""
    75	    billing = await get_billing(session, tenant_id, billing_id)
    76	    _require_postable(billing, billing_id)
    77	
    78	    # Resolve the AR control + revenue accounts UP FRONT (downward reads) — a missing posting
    79	    # default
    80	    # raises 422 here, so the post fails before any state change (the match-post precedent, D-046).
    81	    ar_account_id = await finance_queries.ar_control_account(session, tenant_id)
    82	    revenue_account_id = await finance_queries.sales_revenue_account(session, tenant_id)
    83	
    84	    lines = await get_billing_lines(session, tenant_id, billing_id)
    85	    order_lines = {
    86	        line.id: line
    87	        for line in await get_sales_order_lines(session, tenant_id, billing.sales_order_id)
    88	    }
    89	    invoice_lines = _raise_invoiced_and_build_invoice_lines(lines, order_lines)
    90	    await _advance_order_status(
    91	        session, tenant_id, billing.sales_order_id, order_lines.values()
    92	    )
    93	
    94	    customer = await sales_queries.get_customer(session, tenant_id, billing.customer_id)
    95	    partner_name = customer.name if customer is not None else ""
    96	    due_date = billing.billing_date + timedelta(days=billing.payment_terms_days)
    97	
    98	    billing.status = BillingStatus.POSTED.value
    99	    billing.posted_at = datetime.now()
   100	    await session.flush()
   101	    await docflow.set_document_status(
   102	        session, tenant_id, billing.document_id, status=BillingStatus.POSTED.value
   103	    )
   104	    order = await session.get(SalesOrder, billing.sales_order_id)
   105	    if order is not None:
   106	        await docflow.link_documents(
   107	            session,
   108	            tenant_id,
   109	            predecessor=order.document_id,
   110	            successor=billing.document_id,
   111	            link_type=ORDER_BILLED_BY_BILLING_LINK,
   112	        )
   113	
   114	    # Publish AFTER sales' own writes settle: finance's handler creates + posts the AR customer
   115	    # invoice, drained in this same uow (D-011/D-046).
   116	    publish(
   117	        session,
   118	        BillingInvoiced(
   119	            tenant_id=tenant_id,
   120	            billing_id=billing.id,
   121	            billing_number=billing.billing_number,
   122	            document_id=billing.document_id,
   123	            partner_id=billing.customer_id,
   124	            partner_name=partner_name,
   125	            billing_date=billing.billing_date,
   126	            due_date=due_date,
   127	            currency_code=billing.currency_code,
   128	            ar_account_id=ar_account_id,
   129	            revenue_account_id=revenue_account_id,
   130	            lines=tuple(invoice_lines),
   131	        ),
   132	    )
   133	    return billing

   136	def _raise_invoiced_and_build_invoice_lines(
   137	    lines: list[SalesBillingLine],
   138	    order_lines: dict[uuid.UUID, SalesOrderLine],
   139	) -> list[BillingInvoiceLine]:
   140	    """Raise each order line's invoiced_quantity by its billed quantity and build the per-line AR
   141	    invoice payload finance's handler posts (net + tax code, the item as the dimension). The order
   142	    lines are mutated in-session."""
   143	    invoice_lines: list[BillingInvoiceLine] = []
   144	    for line in lines:
   145	        order_line = order_lines[line.sales_order_line_id]
   146	        new_invoiced = Decimal(str(order_line.invoiced_quantity)) + Decimal(
   147	            str(line.quantity)
   148	        )
   149	        # #75: the create-path check reads the persisted column, which another document may
   150	        # have raised since this DRAFT was created — re-check the cap at post time.
   151	        if new_invoiced > Decimal(str(order_line.delivered_quantity)):
   152	            raise ValidationFailedError(
   153	                message="Posting this billing would exceed the order line's delivered quantity",
   154	                code="sales.over_billing",
   155	                details={
   156	                    "sales_order_line_id": str(order_line.id),
   157	                    "delivered_quantity": str(order_line.delivered_quantity),
   158	                    "invoiced_quantity": str(order_line.invoiced_quantity),
   159	                    "quantity": str(line.quantity),
   160	                },
   161	            )
   162	        order_line.invoiced_quantity = new_invoiced
   163	        invoice_lines.append(
   164	            BillingInvoiceLine(
   165	                item_id=line.item_id,
   166	                net_amount=Decimal(str(line.line_amount)),
   167	                tax_code_id=line.tax_code_id,

   169	        )
   170	    return invoice_lines

   173	async def _advance_order_status(
   174	    session: AsyncSession,
   175	    tenant_id: uuid.UUID,
   176	    order_id: uuid.UUID,
   177	    order_lines,
   178	) -> None:
   179	    """Advance the order after a billing raised the lines' invoiced_quantity (PLAN 7.4): CLOSED when
   180	    every line is fully delivered AND fully invoiced (the O2C end state), else INVOICED when every
   181	    line is fully invoiced (but not yet fully delivered — more to ship), else leave the
   182	    delivery-driven status (PARTIALLY_DELIVERED / DELIVERED) untouched (more to invoice). The order
   183	    +
   184	    lines are already loaded/mutated in this session."""
   185	    fully_invoiced = all(
   186	        Decimal(str(line.invoiced_quantity)) >= Decimal(str(line.ordered_quantity))
   187	        for line in order_lines
   188	    )
   189	    fully_delivered = all(
   190	        Decimal(str(line.delivered_quantity)) >= Decimal(str(line.ordered_quantity))
   191	        for line in order_lines
   192	    )
   193	    if not fully_invoiced:
   194	        return
   195	    new_status = (
   196	        SalesOrderStatus.CLOSED if fully_delivered else SalesOrderStatus.INVOICED
   197	    )
   198	    order = await get_sales_order(session, tenant_id, order_id)
   199	    if order is None:
   200	        return
   201	    order.status = new_status.value
   202	    await session.flush()
   203	    await docflow.set_document_status(
   204	        session, tenant_id, order.document_id, status=new_status.value
   205	    )
